// config is read from the environment, following twelve-factor practice.
type config struct {
        // Addr is the listen address (ADDR, default ":8080").
        Addr string
        // StaticDir, if set, is a directory of frontend files to serve (STATIC_DIR).
        StaticDir string
        // ShutdownTimeout bounds graceful shutdown (SHUTDOWN_TIMEOUT, default 10s).
        ShutdownTimeout time.Duration
}

// loadConfig takes getenv as a parameter so tests need not touch the real
// environment. It fails fast on invalid values instead of starting a server
// that cannot do its job.
func loadConfig(getenv func(string) string) (config, error) {
        cfg := config{
                Addr:            ":8080",
                StaticDir:       getenv("STATIC_DIR"),
                ShutdownTimeout: 10 * time.Second,
        }

        if v := getenv("ADDR"); v != "" {
                cfg.Addr = v
        }

        if v := getenv("SHUTDOWN_TIMEOUT"); v != "" {
                d, err := time.ParseDuration(v)
                if err != nil || d <= 0 {
                        return config{}, fmt.Errorf("SHUTDOWN_TIMEOUT must be a positive duration such as 10s, got %q", v)
                }
                cfg.ShutdownTimeout = d
        }

        if cfg.StaticDir != "" {
                info, err := os.Stat(cfg.StaticDir)
                if err != nil {
                        return config{}, fmt.Errorf("STATIC_DIR: %w", err)
                }
                if !info.IsDir() {
                        return config{}, fmt.Errorf("STATIC_DIR %q is not a directory", cfg.StaticDir)
                }
        }

        return cfg, nil
}

// healthcheck calls the local /healthz endpoint and reports whether it
// answered 204 No Content.
func healthcheck(getenv func(string) string) error {
        cfg, err := loadConfig(getenv)
        if err != nil {
                return err
        }
        url, err := healthURL(cfg.Addr)
        if err != nil {
                return err
        }

        client := &http.Client{Timeout: 2 * time.Second}
        resp, err := client.Get(url) //nolint:gosec // URL is built from our own listen address.
        if err != nil {
                return err
        }
        defer resp.Body.Close()

        if resp.StatusCode != http.StatusNoContent {
                return fmt.Errorf("%s returned %s", url, resp.Status)
        }
        return nil
}

// healthURL maps a listen address such as ":8080" or "0.0.0.0:8080" to a URL
// reachable from inside the same container.
func healthURL(addr string) (string, error) {
        host, port, err := net.SplitHostPort(addr)
        if err != nil {
                return "", fmt.Errorf("invalid ADDR %q: %w", addr, err)
        }
        if host == "" || host == "0.0.0.0" || host == "::" {
                host = "127.0.0.1"
        }
        return "http://" + net.JoinHostPort(host, port) + "/healthz", nil
}

package main

import (
        "context"
        "errors"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/example/calculator/backend/internal/api"
)

func main() {
        logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))

        // "healthcheck" lets the container probe itself; the distroless runtime
        // image has no shell or curl.
        if len(os.Args) > 1 && os.Args[1] == "healthcheck" {
                if err := healthcheck(os.Getenv); err != nil {
                        logger.Error("healthcheck failed", "error", err)
                        os.Exit(1)
                }
                return
        }

        if err := run(logger); err != nil {
                logger.Error("server stopped", "error", err)
                os.Exit(1)
        }
}

func run(logger *slog.Logger) error {
        cfg, err := loadConfig(os.Getenv)
        if err != nil {
                return err
        }

        var static http.Handler
        if cfg.StaticDir != "" {
                static = http.FileServerFS(os.DirFS(cfg.StaticDir))
        }

        srv := &http.Server{
                Addr:              cfg.Addr,
                Handler:           api.NewRouter(logger, static),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        errCh := make(chan error, 1)
        go func() {
                logger.Info("listening", "addr", cfg.Addr, "static_dir", cfg.StaticDir)
                errCh <- srv.ListenAndServe()
        }()

        select {
        case err := <-errCh:
                return err
        case <-ctx.Done():
        }

        logger.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), cfg.ShutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return err
        }
        if err := <-errCh; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

// that knows about HTTP and JSON, and it implements the contract documented
// in backend/api/openapi.yaml.

package api

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "log/slog"

// maxBodyBytes bounds request bodies; a calculation request is tiny.
const maxBodyBytes = 1 << 16

// calculationRequest is the body of POST /api/v1/calculations.
//
// Operands are pointers so that a JSON null can be told apart from 0.
type calculationRequest struct {
        Operation string     `json:"operation"`
        Operands  []*float64 `json:"operands"`
}

// calculationResponse is the success body of POST /api/v1/calculations. The
// client already knows what it asked for, so only the result is returned.
type calculationResponse struct {
        Result float64 `json:"result"`
}

// domainErrors maps calculator errors to HTTP statuses and problem codes.
// Malformed input is a 400; well-formed input that is mathematically
// invalid is a 422.
var domainErrors = []struct {
        err    error
        status int
        code   string
}{
        {calculator.ErrUnsupportedOperation, http.StatusBadRequest, codeUnsupportedOperation},
        {calculator.ErrOperandCount, http.StatusBadRequest, codeInvalidOperandCount},
        {calculator.ErrInvalidOperand, http.StatusBadRequest, codeInvalidOperand},
        {calculator.ErrDivisionByZero, http.StatusUnprocessableEntity, codeDivisionByZero},
        {calculator.ErrNegativeSqrt, http.StatusUnprocessableEntity, codeNegativeSqrt},
        {calculator.ErrUndefinedResult, http.StatusUnprocessableEntity, codeUndefinedResult},
        {calculator.ErrOutOfRange, http.StatusUnprocessableEntity, codeOutOfRange},
}

// NewRouter wires the routes and middleware. When static is non-nil it
// serves every path outside /api/ and /healthz, which is how the production
// container ships the frontend and backend as a single process.
func NewRouter(logger *slog.Logger, static http.Handler) http.Handler {
        mux := http.NewServeMux()

        route(mux, http.MethodPost, "/api/v1/calculations", createCalculation)
        route(mux, http.MethodGet, "/healthz", health)
        mux.HandleFunc("/api/", notFound)
        if static != nil {
                mux.Handle("/", static)
        }

        return chain(mux,
                requestID,
                logRequests(logger),
                recoverPanics(logger),
                securityHeaders,
        )
}

// route registers h for method on path, and a Problem Details 405 for every
// other method on the same path (the default ServeMux 405 is plain text).
func route(mux *http.ServeMux, method, path string, h http.HandlerFunc) {
        allow := method
        if method == http.MethodGet {
                allow = "GET, HEAD"
        }
        mux.HandleFunc(method+" "+path, h)
        mux.HandleFunc(path, func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Allow", allow)
                writeProblem(w, http.StatusMethodNotAllowed, codeMethodNotAllowed,
                        fmt.Sprintf("Method %s is not allowed on this resource; allowed: %s.", r.Method, allow))
        })
}

func createCalculation(w http.ResponseWriter, r *http.Request) {
        if !isJSON(r.Header.Get("Content-Type")) {
                writeProblem(w, http.StatusUnsupportedMediaType, codeUnsupportedMediaType,
                        "Content-Type must be application/json.")
                return
        }

        req, err := decodeCalculationRequest(w, r)
        if err != nil {
                var tooLarge *http.MaxBytesError
                if errors.As(err, &tooLarge) {
                        writeProblem(w, http.StatusRequestEntityTooLarge, codePayloadTooLarge,
                                fmt.Sprintf("Request body must not exceed %d bytes.", maxBodyBytes))
                        return
                }
                writeProblem(w, http.StatusBadRequest, codeInvalidRequest, sentence(err.Error()))
                return
        }

        operands, err := derefOperands(req.Operands)
        if err != nil {
                writeProblem(w, http.StatusBadRequest, codeInvalidOperand, sentence(err.Error()))
                return
        }

        result, err := calculator.Calculate(calculator.Operation(req.Operation), operands)
        if err != nil {
                writeCalculationError(w, err)
                return
        }

        // 200 rather than 201: the calculation is evaluated, not stored, so
        // there is no new resource to point a Location header at.
        writeJSON(w, http.StatusOK, calculationResponse{Result: result})
}

func health(w http.ResponseWriter, _ *http.Request) {
        w.WriteHeader(http.StatusNoContent)
}

func notFound(w http.ResponseWriter, r *http.Request) {
        writeProblem(w, http.StatusNotFound, codeNotFound, fmt.Sprintf("No resource at %s.", r.URL.Path))
}

func isJSON(contentType string) bool {
        mediaType, _, err := mime.ParseMediaType(contentType)
        return err == nil && (mediaType == "application/json" || strings.HasSuffix(mediaType, "+json"))
}

// decodeCalculationRequest strictly decodes the body: unknown fields,
// trailing data and a missing operation or operand list are all rejected.
func decodeCalculationRequest(w http.ResponseWriter, r *http.Request) (calculationRequest, error) {
        var req calculationRequest

        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&req); err != nil {
                return req, describeDecodeError(err)
        }
        if _, err := dec.Token(); !errors.Is(err, io.EOF) {
                return req, errors.New("request body must contain a single JSON object")
        }

        if req.Operation == "" {
                return req, errors.New(`field "operation" is required`)
        }
        if req.Operands == nil {
                return req, errors.New(`field "operands" is required`)
        }
        return req, nil
}

// describeDecodeError rewrites encoding/json errors, which mention Go types,
// into messages that make sense to an API client.
func describeDecodeError(err error) error {
        var (
                tooLarge  *http.MaxBytesError
                syntaxErr *json.SyntaxError
                typeErr   *json.UnmarshalTypeError
        )
        switch {
        case errors.As(err, &tooLarge):
                return err
        case errors.Is(err, io.EOF):
                return errors.New("request body is empty")
        case errors.Is(err, io.ErrUnexpectedEOF):
                return errors.New("request body is not valid JSON: unexpected end of input")
        case errors.As(err, &syntaxErr):
                return fmt.Errorf("request body is not valid JSON: syntax error at offset %d", syntaxErr.Offset)
        case errors.As(err, &typeErr):
                if typeErr.Field == "" {
                        return errors.New("request body must be a JSON object")
                }
                return fmt.Errorf("field %q must be %s, got %s", typeErr.Field, describeKind(typeErr.Type.Kind()), typeErr.Value)
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                return fmt.Errorf("request body contains unknown field %s", strings.TrimPrefix(err.Error(), "json: unknown field "))
        default:
                return errors.New("request body is not valid JSON")
        }
}

func describeKind(k reflect.Kind) string {
        switch k {
        case reflect.Float64:
                return "a finite number"
        case reflect.String:
                return "a string"
        case reflect.Slice:
                return "an array"
        default:
                return "a valid value"
        }
}

func derefOperands(in []*float64) ([]float64, error) {
        out := make([]float64, len(in))
        for i, v := range in {
                if v == nil {
                        return nil, fmt.Errorf("every operand must be a number, but operands[%d] is null", i)
                }
                out[i] = *v
        }
        return out, nil
}

func writeCalculationError(w http.ResponseWriter, err error) {
        for _, m := range domainErrors {
                if errors.Is(err, m.err) {
                        writeProblem(w, m.status, m.code, sentence(err.Error()))
                        return
                }
        }
        writeProblem(w, http.StatusInternalServerError, codeInternal, "An unexpected error occurred.")
}

// sentence capitalises the first letter and adds a full stop, turning a Go
// error string into user-facing text.
func sentence(s string) string {
        if s == "" {
                return s
        }
        return strings.ToUpper(s[:1]) + s[1:] + "."
}

func writeJSON(w http.ResponseWriter, status int, body any) {
        writeBody(w, "application/json", status, body)
}

func writeBody(w http.ResponseWriter, contentType string, status int, body any) {
        w.Header().Set("Content-Type", contentType)
        w.WriteHeader(status)
        // The status line is already sent, so an encoding error can only be
        // a broken connection; there is nothing useful left to do with it.
        _ = json.NewEncoder(w).Encode(body)
}

type middleware func(http.Handler) http.Handler

// chain applies middleware so that the first one listed is the outermost.
func chain(h http.Handler, mw ...middleware) http.Handler {
        for i := len(mw) - 1; i >= 0; i-- {
                h = mw[i](h)
        }
        return h
}

type ctxKey struct{}

const requestIDHeader = "X-Request-ID"

// validRequestID limits accepted client IDs so they are safe to log.
var validRequestID = regexp.MustCompile(`^[A-Za-z0-9._-]{1,64}$`)

// requestID propagates a caller-supplied X-Request-ID or generates one, and
// echoes it on the response so client and server logs can be correlated.
func requestID(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                id := r.Header.Get(requestIDHeader)
                if !validRequestID.MatchString(id) {
                        id = newRequestID()
                }
                w.Header().Set(requestIDHeader, id)
                next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), ctxKey{}, id)))
        })
}

func newRequestID() string {
        b := make([]byte, 8)
        _, _ = rand.Read(b) // crypto/rand.Read never returns an error.
        return hex.EncodeToString(b)
}

// requestIDFrom returns the request ID stored by the requestID middleware.
func requestIDFrom(ctx context.Context) string {
        id, _ := ctx.Value(ctxKey{}).(string)
        return id
}

// statusRecorder captures the status code written by a handler.
type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (r *statusRecorder) WriteHeader(status int) {
        r.status = status
        r.ResponseWriter.WriteHeader(status)
}

// logRequests logs one structured line per request.
func logRequests(logger *slog.Logger) middleware {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}

                        next.ServeHTTP(rec, r)

                        logger.Info("request",
                                "request_id", requestIDFrom(r.Context()),
                                "method", r.Method,
                                "path", r.URL.Path,
                                "status", rec.status,
                                "duration", time.Since(start),
                        )
                })
        }
}

// recoverPanics turns a handler panic into a 500 Problem response instead of
// a dropped connection, and logs it with the request ID.
func recoverPanics(logger *slog.Logger) middleware {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        defer func() {
                                v := recover()
                                if v == nil {
                                        return
                                }
                                // ErrAbortHandler is net/http's sanctioned way to abort a
                                // response; let the server handle it as designed.
                                if err, ok := v.(error); ok && errors.Is(err, http.ErrAbortHandler) {
                                        panic(v)
                                }
                                logger.Error("panic serving request",
                                        "request_id", requestIDFrom(r.Context()),
                                        "panic", v,
                                )
                                writeProblem(w, http.StatusInternalServerError, codeInternal, "An unexpected error occurred.")
                        }()
                        next.ServeHTTP(w, r)
                })
        }
}

// securityHeaders sets conservative defaults for every response. API
// responses are computed per request, so they must not be cached.
func securityHeaders(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                h := w.Header()
                h.Set("X-Content-Type-Options", "nosniff")
                h.Set("Referrer-Policy", "no-referrer")
                h.Set("Content-Security-Policy", "default-src 'self'; frame-ancestors 'none'; base-uri 'self'; form-action 'self'")
                if strings.HasPrefix(r.URL.Path, "/api/") {
                        h.Set("Cache-Control", "no-store")
                }
                next.ServeHTTP(w, r)
        })
}

// Problem codes: a stable, machine-readable reason for every error. They are
// part of the API contract, so clients branch on them, never on detail text.
const (
        codeInvalidRequest       = "INVALID_REQUEST"
        codeUnsupportedMediaType = "UNSUPPORTED_MEDIA_TYPE"
        codePayloadTooLarge      = "PAYLOAD_TOO_LARGE"
        codeUnsupportedOperation = "UNSUPPORTED_OPERATION"
        codeInvalidOperandCount  = "INVALID_OPERAND_COUNT"
        codeInvalidOperand       = "INVALID_OPERAND"
        codeDivisionByZero       = "DIVISION_BY_ZERO"
        codeNegativeSqrt         = "NEGATIVE_SQUARE_ROOT"
        codeUndefinedResult      = "UNDEFINED_RESULT"
        codeOutOfRange           = "RESULT_OUT_OF_RANGE"
        codeNotFound             = "NOT_FOUND"
        codeMethodNotAllowed     = "METHOD_NOT_ALLOWED"
        codeInternal             = "INTERNAL_ERROR"
)

const problemContentType = "application/problem+json"

// problem is an RFC 9457 Problem Details object carrying only what clients
// need. Every member of the RFC is optional: "status" would repeat the HTTP
// status code, "type" defaults to "about:blank", and "title"/"instance" would
// repeat the status text and the request path. Code is an extension member.
type problem struct {
        Code   string `json:"code"`
        Detail string `json:"detail"`
}

func writeProblem(w http.ResponseWriter, status int, code, detail string) {
        writeBody(w, problemContentType, status, problem{Code: code, Detail: detail})
}

// Package calculator implements the arithmetic operations offered by the
// service. It is pure domain logic with no knowledge of HTTP or JSON, which
// keeps it trivial to test and reuse.
package calculator

import (
        "errors"
        "fmt"
        "math"
)

// Operation identifies a supported arithmetic operation.
type Operation string

// Supported operations.
const (
        Add        Operation = "add"
        Subtract   Operation = "subtract"
        Multiply   Operation = "multiply"
        Divide     Operation = "divide"
        Power      Operation = "power"
        Sqrt       Operation = "sqrt"
        Percentage Operation = "percentage"
)

// Errors returned by Calculate. Callers should match them with errors.Is,
// because they may be wrapped with additional context.
var (
        ErrUnsupportedOperation = errors.New("unsupported operation")
        ErrOperandCount         = errors.New("wrong number of operands")
        ErrInvalidOperand       = errors.New("operands must be finite numbers")
        ErrDivisionByZero       = errors.New("division by zero")
        ErrNegativeSqrt         = errors.New("square root of a negative number is not a real number")
        ErrUndefinedResult      = errors.New("result is undefined")
        ErrOutOfRange           = errors.New("result is out of range")
)

type definition struct {
        arity int
        apply func(x []float64) (float64, error)
}

var definitions = map[Operation]definition{
        Add: {2, func(x []float64) (float64, error) {
                return x[0] + x[1], nil
        }},
        Subtract: {2, func(x []float64) (float64, error) {
                return x[0] - x[1], nil
        }},
        Multiply: {2, func(x []float64) (float64, error) {
                return x[0] * x[1], nil
        }},
        Divide: {2, func(x []float64) (float64, error) {
                if x[1] == 0 {
                        return 0, ErrDivisionByZero
                }
                return x[0] / x[1], nil
        }},
        Power: {2, func(x []float64) (float64, error) {
                // 0 to a negative power is 1/0; report it as such rather than as an
                // overflow.
                if x[0] == 0 && x[1] < 0 {
                        return 0, ErrDivisionByZero
                }
                return math.Pow(x[0], x[1]), nil
        }},
        Sqrt: {1, func(x []float64) (float64, error) {
                if x[0] < 0 {
                        return 0, ErrNegativeSqrt
                }
                return math.Sqrt(x[0]), nil
        }},
        // Percentage is "a percent of b".
        Percentage: {2, func(x []float64) (float64, error) {
                return x[0] / 100 * x[1], nil
        }},
}

// Calculate applies op to operands. Operands must be finite and their count
// must match the operation's arity. A result that is not a finite number is
// reported as an error instead of being returned as NaN or ±Inf, which JSON
// cannot represent.
func Calculate(op Operation, operands []float64) (float64, error) {
        def, ok := definitions[op]
        if !ok {
                return 0, fmt.Errorf("%w: %q", ErrUnsupportedOperation, op)
        }
        if len(operands) != def.arity {
                return 0, fmt.Errorf("%w: %s expects %d, got %d", ErrOperandCount, op, def.arity, len(operands))
        }
        for _, v := range operands {
                if math.IsNaN(v) || math.IsInf(v, 0) {
                        return 0, ErrInvalidOperand
                }
        }

        result, err := def.apply(operands)
        if err != nil {
                return 0, err
        }
        switch {
        case math.IsNaN(result):
                return 0, ErrUndefinedResult
        case math.IsInf(result, 0):
                return 0, ErrOutOfRange
        case result == 0:
                // Normalise -0 so clients never see "-0".
                return 0, nil
        }
        return result, nil
}
